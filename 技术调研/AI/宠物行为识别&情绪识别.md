### 2.3 推荐技术栈

#### 方案A: YOLO + 行为分类（推荐）

```
摄像头画面 → YOLO检测宠物 → 裁剪ROI → 行为分类模型 → 输出行为类别
```

**技术选型**:
- 检测: YOLOv8/YOLO12（120种犬类检测）
- 分类: ResNet / EfficientNet / Vision Transformer
- 时序: LSTM / Transformer（捕捉行为序列）

**优点**:
- YOLO实时性好，适合家用摄像头
- 成熟生态，部署简单
- 可基于开源数据集快速训练

---

#### 方案B: DeepLabCut + 行为分类

```
摄像头画面 → DeepLabCut姿态估计 → 骨架序列 → 行为分类 → 输出行为类别
```

**技术选型**:
- 姿态估计: DeepLabCut
- 行为分类: HMM / LSTM / 1D-CNN

**优点**:
- 姿态信息更丰富，行为识别更精准
- 对遮挡、光照变化更鲁棒
- 科研验证，精度高

**缺点**:
- 需要标注姿态数据
- 计算量较大

### 7.2 阶段一：快速原型（1-2周）

**目标**: 验证技术可行性

**步骤**:
1. 使用 **SuperAnimal-Quadruped** 直接对视频进行姿态估计
2. 基于骨架特征手动设计规则分类行为（如：静止时长>5min=睡觉）
3. 验证姿态估计精度和行为分类效果

**代码示例**:
```python
# 使用DeepLabCut SuperAnimal
import deeplabcut

# 加载预训练模型
model = deeplabcut.load_model("superanimal_quadruped")

# 对视频进行姿态估计
keypoints = model.predict_video("pet_video.mp4")

# 简单规则分类
if is_lying_down(keypoints) and duration > 300:  # 5分钟
    behavior = "sleeping"
```

### 7.3 阶段二：模型训练（2-4周）

**目标**: 训练自定义行为分类模型

**数据准备**:
1. 收集自有宠物视频（每种行为100-500段）
2. 使用SuperAnimal提取骨架序列
3. 标注行为类别

**模型训练**:
```python
# 方案1: 基于骨架的LSTM分类器
import torch
import torch.nn as nn

class BehaviorLSTM(nn.Module):
    def __init__(self, input_size=34, hidden_size=128, num_classes=5):
        super().__init__()
        self.lstm = nn.LSTM(input_size, hidden_size, batch_first=True)
        self.fc = nn.Linear(hidden_size, num_classes)

    def forward(self, x):
        # x: (batch, seq_len, 34) - 17个关键点 * 2坐标
        out, _ = self.lstm(x)
        return self.fc(out[:, -1, :])

# 方案2: YOLOv8-Pose + 分类头
from ultralytics import YOLO

model = YOLO("yolov8n-pose.pt")
model.train(data="pet_pose.yaml", epochs=100)
```

### 8.5 姿态分类逻辑

```python
import numpy as np

class PoseClassifier:
    """基于关键点相对位置的姿态分类器"""

    def __init__(self, keypoints):
        """
        keypoints: np.array, shape (20, 3) - x, y, visibility
        """
        self.kp = keypoints
        self.extract_key_points()

    def extract_key_points(self):
        """提取各部位关键点"""
        # 头部
        self.nose = self.kp[2][:2]
        self.throat = self.kp[5][:2]

        # 躯干
        self.withers = self.kp[6][:2]   # 背部前端
        self.croup = self.kp[8][:2]     # 臀部

        # 四肢
        self.front_paws = [self.kp[13][:2], self.kp[14][:2]]  # 前爪
        self.hind_paws = [self.kp[19][:2], self.kp[20][:2]]   # 后爪

    def get_body_metrics(self):
        """计算身体度量"""
        # 身体高度: 喉咙到地面的距离
        all_paws_y = [p[1] for p in self.front_paws + self.hind_paws]
        ground_y = max(all_paws_y)  # Y轴向下，最大值为地面
        body_height = ground_y - self.throat[1]

        # 身体长度: 背部前端到臀部的距离
        body_length = np.linalg.norm(
            np.array(self.withers) - np.array(self.croup)
        )

        return body_height, body_length

    def classify(self):
        """分类姿态"""
        body_height, body_length = self.get_body_metrics()

        # 防止除零
        if body_length < 1e-6:
            return "unknown"

        height_ratio = body_height / body_length

        # 姿态判断规则
        if height_ratio < 0.3:
            return "lying_down"    # 趴着
        elif height_ratio < 0.5:
            return "sitting"       # 坐着
        elif height_ratio < 0.7:
            return "standing"      # 站立
        else:
            return "standing_tall" # 昂首站立


class BehaviorClassifier:
    """基于时序姿态的行为分类器"""

    def __init__(self, fps=30):
        self.fps = fps
        self.pose_history = []

    def update(self, keypoints):
        """更新姿态历史"""
        pose = PoseClassifier(keypoints).classify()
        self.pose_history.append(pose)

        # 保留最近30秒的数据
        max_history = 30 * self.fps
        if len(self.pose_history) > max_history:
            self.pose_history = self.pose_history[-max_history:]

    def classify(self, window_seconds=5):
        """
        分类当前行为
        window_seconds: 分析窗口（秒）
        """
        window = int(window_seconds * self.fps)
        recent_poses = self.pose_history[-window:] if self.pose_history else []

        if not recent_poses:
            return "unknown"

        # 统计姿态分布
        pose_counts = {}
        for p in recent_poses:
            pose_counts[p] = pose_counts.get(p, 0) + 1

        dominant_pose = max(pose_counts, key=pose_counts.get)
        dominant_ratio = pose_counts[dominant_pose] / len(recent_poses)

        # 行为判断
        if dominant_pose == "lying_down" and dominant_ratio > 0.9:
            return "sleeping"      # 睡觉
        elif dominant_pose == "lying_down" and dominant_ratio > 0.7:
            return "resting"       # 休息
        elif dominant_pose == "sitting":
            return "sitting"       # 坐着
        elif self._is_eating(recent_poses):
            return "eating"        # 进食
        elif self._is_playing(recent_poses):
            return "playing"       # 玩耍
        elif self._is_walking(recent_poses):
            return "walking"       # 行走
        else:
            return "active"        # 活跃

    def _is_eating(self, poses):
        """检测进食行为（头部低垂+相对静止）"""
        # 需要结合头部位置判断
        return False  # 简化实现

    def _is_playing(self, poses):
        """检测玩耍行为（姿态快速变化）"""
        if len(poses) < 10:
            return False
        changes = sum(1 for i in range(1, len(poses)) if poses[i] != poses[i-1])
        return changes / len(poses) > 0.3

    def _is_walking(self, poses):
        """检测行走行为（站立+缓慢移动）"""
        standing_ratio = sum(1 for p in poses if p == "standing") / len(poses)
        return standing_ratio > 0.7
```

### 8.6 完整推理流程

```python
import cv2
from ultralytics import YOLO

# 加载模型
model = YOLO("pet_pose/yolo11_exp/weights/best.pt")

# 初始化行为分类器
behavior_cls = BehaviorClassifier(fps=30)

# 视频推理
cap = cv2.VideoCapture("pet_video.mp4")

while cap.isOpened():
    ret, frame = cap.read()
    if not ret:
        break

    # YOLO推理
    results = model(frame, verbose=False)

    if results[0].keypoints is not None:
        # 获取关键点 (N, 20, 3)
        keypoints = results[0].keypoints.data[0].cpu().numpy()

        # 更新行为分类器
        behavior_cls.update(keypoints)

        # 获取当前行为
        behavior = behavior_cls.classify()
        print(f"Current behavior: {behavior}")

        # 绘制关键点
        annotated_frame = results[0].plot()
        cv2.putText(annotated_frame, f"Behavior: {behavior}",
                    (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2)
        cv2.imshow("Pet Behavior", annotated_frame)

    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
```
